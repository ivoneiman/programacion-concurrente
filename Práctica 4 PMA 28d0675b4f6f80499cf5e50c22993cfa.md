no podría realizar las tareas administrativas. Entonces si no hay clientes para atender
para no generar demora innecesaria en el recieve debemos crear una especie de proceso
coordinador que manda a los Clientes a los Empleados que pidan atender.

chan Atender(int)
chan Llegada(int)
chan Siguiente[2](int)

Process Cliente [id:0..N-1]{
	//Avisa que llega
	send LLegada(id);
}
Process Cordinador{
	int idE,idC;
	while (true){
		//Se bloquea esperando a que un Empleado se libere
		receive Atender(idE);
		//Si no llego cliente, le aviso con -1
		if (empty(Llegada)){
			idC=-1;
		}
		else{
			receive LLegada(idC)
		}
		send Siguiente[idE](idC);
	}
}

			receive pedidos(idC,Ped);
		}
		send pedidoSiguiente[idV](idC,Ped);
	}

Process Vendedor [id: 0..2]{
	int idC;
	text p;
	while (true){
		receive pedidoSiguiente[id](idC,Ped);
		if (Ped != "VACIO"){
			send platos(idC,Ped);
		}
		else {
			delay(60*3); //Repone la heladera
		}
	}
}

Process Cocinero[2]{
	int idC;
	text plato;
	while (true){
		receive platos(idC,plato);
		//Cocina el plato

	}

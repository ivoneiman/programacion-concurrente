	}

Process Admin{
	text m;

		EmpleadoMuestra?muestra(m) -> push (Buffer,m);

	od

	Sistema!entrar(id);

Process Sistema{
	int idE;
	bool libre=true;
	
		while(cantE<E){
			if
				Espectador[*]?entrar(idE) -> if(libre){
																				libre=false
																				Espectador[idE]!usar();
																			} else push(fila(idE))
				
				[] Espectador[*]?salir() -> if (empty fila){
																				libre = true;
																		} else{
																			pop(fila(idE))
																			Espectador[idE]!usar();
																		}

		}
